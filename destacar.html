ARQUIVO HTML Destacar Nome

destacar.html

Escreva um nome e veja ele destacado na lista

Se o nome não estiver na lista, ou não for exatamente igual, ele não será destacado

    ARQUIVO JAVASCRIPT var nomes = [ "João", "Márcio", "Fabiana", "Jorge", "Silvana", "Luiz", "Carlos", "Pablo" ]; function destacar() { var txt = document.getElementById("txtNome"); var lista = document.getElementById("listaNomes"); var busca = txt.value; lista.innerHTML = ""; for (var i = 0; i < nomes.length; i++) { if (nomes[i] == busca) { lista.innerHTML += `
  1. ${nomes[i]}
    2. `; } else { lista.innerHTML += `
  2. ${nomes[i]}
    3. `; } } } destacar();
