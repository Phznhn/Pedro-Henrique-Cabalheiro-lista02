ARQUIVO JAVA SCRIPT function verificar() { var data = new Date(); var anoAtual = data.getFullYear(); var fano = document.getElementById("txtano"); var res = document.getElementById("res"); var img = document.getElementById("foto"); var anoNasc = Number(fano.value); var idade = anoAtual - anoNasc; var fsex = document.getElementsByName("radsex"); var genero = ""; if (fsex[0].checked) { genero = "Homem"; if (idade >= 0 && idade <= 2) { img.src = "bebe menino.webp"; } else if (idade < 13) { img.src = "crianca menino.webp"; } else if (idade < 18) { img.src = "adolescente menino.webp"; } else if (idade < 60) { img.src = "adulto homem.webp"; } else { img.src = "idoso homem.webp"; } } else { genero = "Mulher"; if (idade >= 0 && idade <= 2) { img.src = "bebe menina.webp"; } else if (idade < 13) { img.src = "crianca menina.webp"; } else if (idade < 18) { img.src = "adolescente menina.webp"; } else if (idade < 60) { img.src = "adulto mulher.webp"; } else { img.src = "idoso mulher.webp"; } } res.innerHTML = `Detectamos ${genero.toLowerCase()} com ${idade} anos.`; img.style.display = "inline-block"; }
