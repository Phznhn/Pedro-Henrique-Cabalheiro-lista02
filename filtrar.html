Crie um campo de entrada que filtre os itens de uma lista conforme o usuário digita.

    ARQUIVO JAVA SCRIPT var nomes = [ "João", "Márcio", "Fabiana", "Jorge", "Silvana", "Luiz", "Carlos", "Pablo" ]; function filtrarNomes() { var campo = document.getElementById("campoBusca"); var lista = document.getElementById("listaNomes"); var texto = campo.value; lista.innerHTML = ""; for (var i = 0; i < nomes.length; i++) { if (nomes[i].indexOf(texto) != -1) { lista.innerHTML += `
  • ${nomes[i]}
    • `; } } } filtrarNomes();
